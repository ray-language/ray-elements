<script type="module">
  const p = document.getElementById("demo-pagination");
  p.addEventListener("change", () => {
    document.getElementById("demo-pagination-out").textContent = `Page ${p.value} of 20`;
  });
</script>
import type { CSSProperties } from "react";
import { RayPagination } from "@ray-ds/react";

// The live example's <script>/<style> is left out: pass data as props
// and listen with the on* handlers (onChange…).
export function PaginationExample() {
  return (
    <>
      <div style={{ display: "grid", gap: "1rem" }}>
        <RayPagination id="demo-pagination" total={20} value={10} />
        <RayPagination total={5} value={2} color="teal" radius="full" size="xs" />
        <RayPagination total={50} value={1} siblings={2} withEdges />
        <RayPagination total={8} value={3} disabled />
      </div>
      <p id="demo-pagination-out" style={{ margin: "0.75rem 0 0", color: "var(--ray-muted-foreground)", fontSize: "var(--ray-font-size-sm)" } as CSSProperties}>Page 10 of 20</p>
    </>
  );
}
<!-- app.use(RayDs) (from "@ray-ds/vue") registers the components. -->
<!-- The live example's <script>/<style> is left out: bind data with :prop and listen with @event. -->
<template>
  <div style="display: grid; gap: 1rem">
    <RayPagination id="demo-pagination" :total="20" :value="10" />
    <RayPagination :total="5" :value="2" color="teal" radius="full" size="xs" />
    <RayPagination :total="50" :value="1" :siblings="2" with-edges />
    <RayPagination :total="8" :value="3" disabled />
  </div>
  <p id="demo-pagination-out" style="margin: 0.75rem 0 0; color: var(--ray-muted-foreground); font-size: var(--ray-font-size-sm)">Page 10 of 20</p>
</template>
