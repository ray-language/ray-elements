<script type="module">
  const shell = document.getElementById("demo-shell");
  const device = document.getElementById("demo-device");
  const nav = document.getElementById("demo-shell-nav");
  const side = document.getElementById("demo-shell-side");
  const bar = document.getElementById("demo-shell-bar");
  const titles = { home: "Home", search: "Search", messages: "Messages", profile: "Profile" };

  // Frame width: the shell measures itself and picks its layout.
  document.getElementById("demo-shell-width").addEventListener("change", (e) => { device.dataset.size = e.target.value; });
  const mode = () => {
    document.getElementById("demo-shell-mode").textContent = shell.mobile
      ? "phone layout: app bar, content, bottom nav"
      : "wide layout: header, sidebar, content, aside";
  };
  shell.addEventListener("ray-breakpoint", mode);
  customElements.whenDefined("ray-app-shell").then(() => requestAnimationFrame(mode));

  // One destination, shown by the bottom nav, the sidebar and the bar.
  const go = (page) => {
    for (const s of shell.querySelectorAll("section[data-page]")) s.hidden = s.dataset.page !== page;
    for (const l of side.querySelectorAll("ray-nav-link")) l.active = l.dataset.page === page;
    nav.value = page;
    bar.label = titles[page];
    shell.scroller.scrollTop = 0;
  };
  nav.addEventListener("change", () => go(nav.value));
  side.addEventListener("click", (e) => { const link = e.target.closest("ray-nav-link"); if (link) go(link.dataset.page); });
  for (const b of shell.querySelectorAll("[data-go]")) b.addEventListener("click", () => go(b.dataset.go));

  // The menu: the drawer on a phone, the sidebar's rail on wide screens.
  document.getElementById("demo-shell-menu").addEventListener("click", () => {
    if (shell.mobile) document.getElementById("demo-shell-drawer").show();
    else side.toggle();
  });
  nav.addEventListener("ray-select", () => document.getElementById("demo-shell-new").show());

  // Content to scroll: the bar elevates, the bottom nav hides going down.
  const builds = shell.querySelector("[data-builds]");
  const states = [["green", "passed"], ["green", "passed"], ["orange", "slow"], ["green", "passed"], ["red", "failed"]];
  for (let i = 0; i < 14; i++) {
    const [color, state] = states[i % states.length];
    const li = document.createElement("li");
    li.innerHTML = `<span class="ic" style="--c: var(--ray-${color}-light-color); --b: var(--ray-${color}-light)"><ray-icon name="terminal"></ray-icon></span><span><b>#${482 - i} ${state}</b><small>main · ${(3 + (i % 4) * 0.7).toFixed(1)} s</small></span><small>${i * 7 + 2} min</small>`;
    builds.append(li);
  }
  const results = shell.querySelector("[data-results]");
  const all = ["#482 passed", "#481 passed", "#480 slow", "Ada Lovelace", "Grace Hopper", "release-notes.md", "bundles/"];
  const search = document.getElementById("demo-shell-search");
  const find = () => {
    const q = (search.value ?? "").toLowerCase();
    results.replaceChildren(...all.filter((r) => r.toLowerCase().includes(q)).map((r) => { const li = document.createElement("li"); li.innerHTML = "<span><b></b></span>"; li.querySelector("b").textContent = r; return li; }));
  };
  search.addEventListener("input", find);
  find();
</script>
